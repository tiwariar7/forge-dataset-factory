API Reference

Full class, method, and signature reference for the ForgeDataset package.

IngestionRouter(config=None)

Inspects document file signatures and routes to optimal extractors (Docling, OpenPyXL, native markdown/CSV).

from ForgeDataset import IngestionRouter

router = IngestionRouter()
doc = router.route("path/to/file.pdf") -> CanonicalDocument

HybridChunker(max_source_context=600, overlap=0, ...)

Performs hierarchical heading-aware chunking with zero token overlap and table header preservation.

from ForgeDataset import HybridChunker

chunker = HybridChunker(
    max_source_context=600,    # Max token budget per chunk
    min_merge_size=150,        # Minimum size before merging peer elements
    merge_peers=True,          # Merge small adjacent elements under same heading
    repeat_table_headers=True, # Repeat table headers across splits
    overlap=0                  # Zero token overlap recommended for SFT
)
chunks = chunker.chunk_document(doc) -> List[Chunk]

SFTGenerator(engine="ollama", model=None, **kwargs)

Orchestrates multi-category SFT generation with verbatim evidence extraction.

from ForgeDataset import SFTGenerator

# With Ollama
gen = SFTGenerator(engine="ollama", model="qwen2.5:7b-instruct")

# With in-process llama.cpp
gen = SFTGenerator(engine="llama_cpp", model_path="qwen.gguf")

# Generate pairs
pairs = gen.generate_from_chunk(chunk) -> List[QAPair]

QualityGate(config=None)

Evaluates candidate pairs for character-level verbatim evidence, numeric validity, and semantic uniqueness.

from ForgeDataset import QualityGate

gate = QualityGate()
evaluated_qa = gate.evaluate(qa_pair, chunk) -> QAPair
# evaluated_qa.status in ['approved', 'needs_review', 'rejected']

DatasetExporter(output_dir="output")

Exports fine-tuning datasets into Alpaca, ShareGPT, and Audit Provenance formats with document-level splits.

from ForgeDataset import DatasetExporter

exporter = DatasetExporter(output_dir="output")
res = exporter.export(
    approved_pairs,
    format_type="alpaca", # 'alpaca' | 'sharegpt' | 'raw_jsonl'
    prefix="dataset",
    train_ratio=0.8,
    val_ratio=0.1,
    test_ratio=0.1
) -> Dict[str, Any]
